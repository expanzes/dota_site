// (Замени функцию loadProfile и ranks внутри script в profile.html) async function loadProfile() { const res = await fetch('/api/me'); const user = await res.json(); if (!user.logged_in) { window.location.href = '/'; return; } userSiteId = user.site_id; document.getElementById('display-username').innerText = user.username; document.getElementById('user-display-name').innerText = user.username; document.getElementById('display-site-id').innerText = user.site_id; document.getElementById('display-score').innerText = user.invoker_score; if (user.avatar) document.getElementById('profile-avatar').src = user.avatar; if (user.is_premium) document.getElementById('premium-badge').classList.remove('hidden'); if (user.steam_linked) document.getElementById('steam-link-btn').classList.add('hidden'); // КРАСИВАЯ РАСШИФРОВКА РАНГА const rankNames = ["Unranked", "Herald", "Guardian", "Crusader", "Archon", "Legend", "Ancient", "Divine", "Immortal"]; let rankText = "Unknown"; if (user.rank && user.rank > 0) { const medalIndex = Math.floor(user.rank / 10); const stars = user.rank % 10; const medalName = rankNames[medalIndex] || "Unknown"; if (medalIndex === 8) { // Immortal rankText = "Immortal"; } else { rankText = `${medalName} [${stars}]`; // Например: Archon [3] } } else { rankText = user.steam_linked ? "Hidden/Unranked" : "Link Steam first"; } document.getElementById('display-rank').innerText = rankText; }
